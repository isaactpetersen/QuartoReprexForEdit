set.seed(52242)
Oxboys_addNA[sample(1:nrow(Oxboys_addNA), 25), "height"] <- NA

dataToImpute <- Oxboys_addNA

head(dataToImpute)

# Imputation settings
numImputations <- 5

varsToImpute <- c("height")

pred[varsToImpute, "Occasion"] <- 1 # fixed effect predictor
pred[varsToImpute, "age"] <- 2 # random effect predictor
