# Quarto Reprex (via GitHub Actions on Linux) {#sec-quarto}

The following reprex is from Quarto (via GitHub Actions on Linux):

```{r}
# Load libraries
library("mice")

# Impute data
imp <- mice::mice(
  nhanes,
  maxit = 2,
  m = 2)

# Fit regression model to multiply imputed data
fit <- with(
  data = imp,
  exp = lm(bmi ~ hyp + chl))

# Pool results across models
fit_pooled <- mice::pool(fit)

# Summarize pooled results
fit_pooled

rmarkdown::paged_table(fit_pooled)

knitr::kable(fit_pooled)

summary(fit_pooled)
```
